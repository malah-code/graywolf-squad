## 2. Relational Schema (PostgreSQL 16 + pgvector)

Canonical schema unifying all table definitions across architecture docs.
Every table containing user data is scoped by `org_id` / `workspace_id` composite tenancy keys with PostgreSQL Row-Level Security (RLS) enabled.

> **Vector Dimension**: All `vector(768)` columns use the `nomic-embed-text` embedding model (Apache 2.0, 768-dimensional output, run locally via Ollama at $0 cost). See ADR-023.

```sql
CREATE EXTENSION IF NOT EXISTS vector;
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- ============================================================
-- TENANCY HIERARCHY (ADR-015, multi-tenancy-and-deployments.md)
-- ============================================================

-- 1. Organizations (Top-level billing & governance boundary)
CREATE TABLE organizations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    slug VARCHAR(64) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    plan_tier VARCHAR(32) NOT NULL DEFAULT 'community', -- 'community' | 'team' | 'enterprise'
    mode VARCHAR(32) NOT NULL DEFAULT 'personal',       -- 'personal' | 'team'
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2. Workspaces (Sub-compartments within an organization)
CREATE TABLE workspaces (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    org_id UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    slug VARCHAR(64) NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(org_id, slug)
);

-- 3. Users & Memberships
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE NOT NULL,
    full_name VARCHAR(255),
    avatar_url TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE workspace_members (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    role VARCHAR(32) NOT NULL DEFAULT 'developer', -- 'owner' | 'lead_dev' | 'developer' | 'viewer'
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(workspace_id, user_id)
);

-- ============================================================
-- CORE ENTITIES
-- ============================================================

-- 4. Projects (Bound to Workspace & Git Remote)
CREATE TABLE projects (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    slug VARCHAR(64) NOT NULL,
    mode VARCHAR(32) NOT NULL DEFAULT 'brownfield', -- 'greenfield' | 'brownfield'
    git_remote_url TEXT NOT NULL,
    git_provider VARCHAR(32) NOT NULL DEFAULT 'github', -- 'github' | 'gitlab' | 'bitbucket' | 'azure' | 'gitea'
    config_json JSONB NOT NULL DEFAULT '{}', -- Full project.config.json
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(workspace_id, slug)
);

-- 5. Tasks (Scoped by Project, Workspace, and Org)
CREATE TABLE tasks (
    id VARCHAR(64) PRIMARY KEY, -- e.g. 'TASK-101'
    project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    title VARCHAR(512) NOT NULL,
    type VARCHAR(16) NOT NULL CHECK (type IN ('feature', 'bug', 'refactor')),
    status VARCHAR(48) NOT NULL DEFAULT 'BACKLOG',
      -- States: BACKLOG | QUEUED | IN_PROGRESS | BLOCKED | AWAITING_CLARIFICATION
      --         INTERNAL_REVIEW | INTERNAL_REVIEW_FAILED
      --         GUARD_VERIFYING | QA_FIXING | VERIFICATION_FAILED
      --         HUMAN_REVIEW_GATE | MERGED | REJECTED | ESCALATED | UPDATED
    priority VARCHAR(16) NOT NULL DEFAULT 'MEDIUM',
    assigned_tier VARCHAR(16), -- dev-level-1 | dev-level-2 | dev-level-3
    blocked_by TEXT[] DEFAULT '{}',
    writable_scope TEXT[] DEFAULT '{}',
    spend_limit_usd NUMERIC(10, 4) DEFAULT 0.0000,
    current_spend_usd NUMERIC(10, 4) DEFAULT 0.0000,
    internal_review_attempts INT DEFAULT 0,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Task dependency graph (DAG)
CREATE TABLE task_dependencies (
    task_id VARCHAR(64) REFERENCES tasks(id) ON DELETE CASCADE,
    blocked_by_task_id VARCHAR(64) REFERENCES tasks(id) ON DELETE CASCADE,
    PRIMARY KEY (task_id, blocked_by_task_id)
);

-- Task artifacts (patches, briefs, plans, logs, etc.)
CREATE TABLE task_artifacts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    task_id VARCHAR(64) REFERENCES tasks(id) ON DELETE CASCADE,
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    artifact_type VARCHAR(32) NOT NULL,
      -- Types: brief | plan | implementation-summary | review-signoff
      --        escalation-request | patch | log | verification | result
      --        e2e-cache | browser-trace | browser-test-evidence
    authored_by VARCHAR(16), -- dev-level-1 | dev-level-2 | dev-level-3 | orchestrator
    file_path TEXT NOT NULL,
    content_hash VARCHAR(64) NOT NULL,
    embedding vector(768), -- nomic-embed-text (768-dim, Apache 2.0, local Ollama)
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Escalation events
CREATE TABLE escalations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    task_id VARCHAR(64) REFERENCES tasks(id) ON DELETE CASCADE,
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    escalation_type VARCHAR(32) NOT NULL,
      -- Types: boundary-violation | internal-review-failed | qa-loop-exceeded | stuck-detector
    agent_tier VARCHAR(16) NOT NULL,
    emitted_at TIMESTAMPTZ DEFAULT NOW(),
    resolved_at TIMESTAMPTZ,
    resolution VARCHAR(16) -- approved | rejected | re-scoped | cancelled
);

-- ============================================================
-- OBSERVABILITY & COST TRACKING (ADR-017, observability-and-telemetry.md)
-- ============================================================

-- Granular LLM Token Ledger & Analytics
CREATE TABLE llm_calls (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    task_id VARCHAR(64) NOT NULL REFERENCES tasks(id) ON DELETE CASCADE,
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    org_id UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
    persona VARCHAR(32) NOT NULL,   -- 'planner' | 'implementer' | 'reviewer'
    model VARCHAR(128) NOT NULL,    -- 'claude-3-5-sonnet-20241022', 'deepseek-v3'
    provider VARCHAR(64) NOT NULL,  -- 'anthropic', 'openrouter', 'ollama'
    input_tokens INT NOT NULL,
    output_tokens INT NOT NULL,
    cache_read_tokens INT DEFAULT 0,  -- prompt cache hits (e.g. static repo map)
    cache_write_tokens INT DEFAULT 0,
    cost_usd NUMERIC(10, 6) NOT NULL, -- $0.00 for local Ollama runs
    latency_ms INT,
    status VARCHAR(16) NOT NULL DEFAULT 'success', -- 'success' | 'rate_limited' | 'error'
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- System-One Decision Gate Audit Trail (ADR-079)
CREATE TABLE task_gate_decisions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    task_id VARCHAR(64) NOT NULL REFERENCES tasks(id) ON DELETE CASCADE,
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    gate VARCHAR(32) NOT NULL,            -- 'sandbox_tier' | 'gate_policy' | 'relay_tool' | 'two_strike_prescreen'
    question TEXT NOT NULL,
    result JSONB NOT NULL,                -- { choice?: string, score?: string, answer?: boolean }
    confidence NUMERIC(4, 3) NOT NULL,    -- 0.000 to 1.000
    latency_ms INT NOT NULL,              -- 70 to 500 ms typical
    model_backend VARCHAR(64) NOT NULL,   -- 'local-ollama' | 'typesafe-jev-api' | 'openjev-server'
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_task_gate_decisions_task ON task_gate_decisions(task_id);
CREATE INDEX idx_task_gate_decisions_gate ON task_gate_decisions(gate, created_at DESC);

-- ============================================================
-- SECURITY (ADR-018, security-and-secrets.md)
-- ============================================================

-- Confidence Calibration (Gap 9 implementation prerequisite)
-- CALIBRATION IS PER-MODEL. Model ID and Version are not optional metadata, they define the validity domain.
-- Nightly aggregation requirements:
--   1. Minimum sample gate: n >= 30 per (file_type, model) cell before adjusting.
--   2. Maximum adjustment: capped at ±0.15.
--   3. Recency window: 90-day half-life matching the learning engine.
--   4. ERP Visibility: Calibrated confidence MUST surface in the Executive Review Packet.
CREATE TABLE confidence_calibration (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    task_id VARCHAR(64) NOT NULL REFERENCES tasks(id) ON DELETE CASCADE,
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    user_id UUID REFERENCES users(id) ON DELETE SET NULL,
    model_id VARCHAR(64) NOT NULL,        -- e.g. 'deepseek-v3'
    model_version VARCHAR(32) NOT NULL,   -- calibration is PER-MODEL
    file_path TEXT NOT NULL,
    file_language VARCHAR(32),
    file_type VARCHAR(32),
    confidence_score NUMERIC(3, 2) NOT NULL,
    justification TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Envelope-Encrypted Secrets Vault
CREATE TABLE secret_vault (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    secret_name VARCHAR(128) NOT NULL,
    encrypted_value BYTEA NOT NULL,  -- AES-256 encrypted via DEK
    encrypted_dek BYTEA NOT NULL,    -- DEK encrypted via external KEK (~/.graywolf/master.key or KMS)
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(workspace_id, secret_name)
);

-- ============================================================
-- CONTINUOUS LEARNING (ADR-022, continuous-learning-engine.md)
-- ============================================================

-- Project Lessons (Three-Tier Learning Engine)
CREATE TABLE project_lessons (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    org_id UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,

    provenance VARCHAR(32) NOT NULL,
      -- 'human_correction' | 'qa_guard_failure' | 'auto_inferred' | 'postmortem'
    confidence_score NUMERIC(3, 2) NOT NULL, -- 0.10 to 1.00
    category VARCHAR(32) NOT NULL,           -- 'convention' | 'bugfix' | 'anti_pattern' | 'perf'
    trigger_scope TEXT[] NOT NULL,           -- e.g. ['src/modules/billing/**', '*.ts']
    summary TEXT NOT NULL,
    rule_content TEXT NOT NULL,
    embedding vector(768),                   -- nomic-embed-text (768-dim)
    times_applied INT DEFAULT 0,
    times_rejected INT DEFAULT 0,
    status VARCHAR(16) NOT NULL DEFAULT 'active', -- 'active' | 'quarantined' | 'archived'
    last_applied_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

    file_path TEXT NOT NULL,
    line_start INT NOT NULL,
```

## 3. Task State Machine

```
BACKLOG
  │ (User / Chat Agent queues for execution)
  ▼
QUEUED (w/ resolved deps)
  │
  ▼
IN_PROGRESS ──────────────── [Minor change] ──▶ (inject to buffer, stay IN_PROGRESS)
  │                   │
  │                   └──── [Major pivot] ──────▶ container restart → QUEUED
  │
  ├──── [Ambiguity detected] ─────▶ BLOCKED (AWAITING_CLARIFICATION)
  │                                    │
  │                                    └── [Human answers] ──▶ IN_PROGRESS (resume)
  ▼
INTERNAL_REVIEW
  │  (Dev-Level-2 runs test suite)
  ├──── [Sign-off] ───────────────▶ GUARD_VERIFYING
  └──── [Reject × 2] ─────────────▶ INTERNAL_REVIEW_FAILED → HUMAN_REVIEW_GATE

GUARD_VERIFYING
  ├──── [All guards pass] ────────▶ HUMAN_REVIEW_GATE
  └──── [Guard fails] ────────────▶ QA_FIXING (auto-retry, max 3)
                                         └── [Max retries exceeded] ▶ VERIFICATION_FAILED
                                                                          → HUMAN_REVIEW_GATE

HUMAN_REVIEW_GATE
  ├──── [Human approves] ─────────▶ MERGED
  └──── [Human rejects] ──────────▶ REJECTED

[Boundary violation at any stage] ──▶ ESCALATED → HUMAN_REVIEW_GATE
```
